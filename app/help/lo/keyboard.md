ປຸ່ມຕົວອັກສອນເຮັດວຽກເມື່ອຕົວຊີ້ຢູ່ບ່ອນໃດກໍໄດ້ໃນໜ້າ ແລະ ທ່ານບໍ່ໄດ້ພິມໃນຊ່ອງ. ໃນຂະນະທີ່ໜ້າຕ່າງເປີດຢູ່, ປຸ່ມຂອງມັນເອງຈະໃຊ້ໄດ້. ຕົວອັກສອນໃຊ້ໄດ້ທັງກົດ {{key:Shift}} ຫຼື ບໍ່ກົດກໍໄດ້.

## ການເລືອກ ແລະ ແກ້ໄຂ

| ປຸ່ມ ຫຼື ການກະທຳ | ມັນເຮັດຫຍັງ |
|---|---|
| ຄລິກເຄື່ອງໝາຍ (ເຄື່ອງມືເລືອກ) | ເລືອກມັນ ແລະ ເຮັດໃຫ້ລາຍການຂອງມັນເປັນປັດຈຸບັນ. |
| {{key:Shift}} + ຄລິກເຄື່ອງໝາຍ | ເພີ່ມເຂົ້າໃນການເລືອກ ຫຼື ເອົາອອກ. |
| {{key:Mod+A}} | ເລືອກທຸກເຄື່ອງໝາຍໃນແຜ່ນທີ່ເປີດຢູ່. |
| ລາກຕົວຈັບມຸມ | ຍ້າຍຈຸດນັ້ນ ({{key:Shift}} ດູດເກາະເປັນ 45°). ໃຊ້ໄດ້ເມື່ອເລືອກເຄື່ອງໝາຍໜຶ່ງອັນ. |
| {{key:Delete}} ຫຼື {{key:Backspace}} | ລຶບເຄື່ອງໝາຍທີ່ເລືອກ (ເມື່ອທ່ານບໍ່ໄດ້ແຕ້ມ). |
| {{key:Esc}} | ບໍ່ເລືອກຫຍັງ: ລຶບການເລືອກ ແລະ ເສັ້ນກວດ. |
| {{key:Mod+Z}} | ຍ້ອນກັບ. |
| {{key:Mod+Y}} ຫຼື {{key:Mod+Shift+Z}} | ເຮັດຊ້ຳ. |

## ສຳເນົາ ແລະ ວາງ

| ປຸ່ມ | ມັນເຮັດຫຍັງ |
|---|---|
| {{key:Mod+C}} | ສຳເນົາເຄື່ອງໝາຍທີ່ເລືອກ, ພ້ອມລາຍການຂອງມັນ. |
| {{key:Mod+X}} | ຕັດ: ສຳເນົາເຄື່ອງໝາຍທີ່ເລືອກ ແລ້ວລຶບມັນ. |
| {{key:Mod+V}} | ວາງ: ສຳເນົາຈະຕາມຕົວຊີ້; ຄລິກບ່ອນທີ່ມັນຄວນໄປ. |
| {{key:Mod+Shift+V}} | ວາງໃນບ່ອນເດີມ: ຕຳແໜ່ງດຽວກັນໃນແຜ່ນທີ່ເປີດຢູ່. |
| {{key:Mod+D}} | ສຳເນົາຊ້ຳເຄື່ອງໝາຍທີ່ເລືອກ: ສຳເນົາຈະຕາມຕົວຊີ້; ຄລິກບ່ອນທີ່ມັນຄວນໄປ. |
| {{key:Esc}} | ບໍ່ວາງຫຍັງ: ຍົກເລີກການວາງ ຫຼື ສຳເນົາຊ້ຳທີ່ຢູ່ກັບຕົວຊີ້. |

ຕົວອັກສອນດ່ຽວຍັງເລືອກເຄື່ອງມືໄດ້ ({{key:C}} ແມ່ນ {{ui:ນັບ}}, {{key:D}} ແມ່ນ {{ui:ກວດ}}, {{key:X}} ແມ່ນ {{ui:ຕັດຮູ}}, {{key:V}} ແມ່ນ {{ui:ເລືອກ}}, {{key:A}} ແມ່ນ {{ui:ພື້ນທີ່}}); ມີແຕ່ເມື່ອ {{key:Mod}} ຢູ່ກັບພວກມັນເທົ່ານັ້ນທີ່ສຳເນົາ ແລະ ວາງ. ໃນຊ່ອງຂໍ້ຄວາມ ປຸ່ມເຫຼົ່ານີ້ສຳເນົາ ແລະ ວາງຂໍ້ຄວາມຕາມປົກກະຕິ. ເບິ່ງ [ສຳເນົາ, ວາງ ແລະ ສຳເນົາຊ້ຳເຄື່ອງໝາຍ](topic:copy-paste).

:::warning
{{key:Backspace}} ລຶບເຄື່ອງໝາຍທີ່ເລືອກເມື່ອທ່ານບໍ່ໄດ້ແຕ້ມ. ຖ້າເກີດຂຶ້ນໂດຍບັງເອີນ, {{key:Mod+Z}} ຈະເອົາມັນກັບມາ.
:::

| {{key:Esc}} | ຍົກເລີກການຄົ້ນຫາທີ່ກຳລັງເຮັດ, ຫຼື ຖິ້ມການກວດທົບທວນ. |

## ວິທີໃຊ້

| ປຸ່ມ | ມັນເຮັດຫຍັງ |
|---|---|
| {{key:F1}} | ເປີດ ຫຼື ປິດແຜງວິທີໃຊ້ (ແລະ ປຸ່ມ {{ui:ວິທີໃຊ້}} ໃນແຖບແອັບ). ໃຊ້ໄດ້ໃນທຸກເຄື່ອງມື, ເຖິງແມ່ນຕອນພິມໃນຊ່ອງ. |
| {{key:Esc}} | ປິດແຜງວິທີໃຊ້. ໃນຊ່ອງຄົ້ນຫາຂອງມັນ, Esc ທຳອິດຈະລຶບການຄົ້ນຫາ. ໃນຂະນະທີ່ແຜງເປີດຢູ່, Esc ທີ່ກົດຫຼັງຈາກຄລິກແຜ່ນ ຈະປິດແຜງກ່ອນ; ກົດອີກເທື່ອເພື່ອຍົກເລີກໃນ TakeoffCraft. |

:::mac
ໃນແປ້ນພິມ Mac ຫຼາຍອັນ ແຖວເທິງຄວບຄຸມຄວາມສະຫວ່າງ ແລະ ສຽງ. ກົດ fn ພ້ອມ F1, ຫຼື ເປີດ "Use F1, F2, etc. keys as standard function keys" ໃນ System Settings, ພາຍໃຕ້ Keyboard.
:::

## ໝາຍເຫດແພລດຟອມ

:::mac
- {{key:Mod}} ແມ່ນປຸ່ມ Command: {{key:Mod+Z}}, {{key:Mod+S}}, {{key:Mod+O}}. ເຕັມຈໍແມ່ນຂໍ້ຍົກເວັ້ນ: Control+Shift+F.
- ການຄລິກຂວາແມ່ນການຄລິກສອງນິ້ວເທິງແທຣັກແພດ, ເມື່ອເປີດ secondary click ໃນ System Settings, ພາຍໃຕ້ Trackpad. TakeoffCraft ອ່ານພຽງແຕ່ປຸ່ມຮອງ; ການ Control-click ອາດນັບເປັນການຄລິກທຳມະດາ ແລະ ເພີ່ມຈຸດ.
- ການປັດສອງນິ້ວເທິງແທຣັກແພດຈະຊູມແຜ່ນ (ບໍ່ເລື່ອນ). ເພື່ອເລື່ອນ, ກົດ ແລ້ວລາກ, ຫຼື ຖື {{key:Space}} ແລ້ວລາກ.
- Page Up ແລະ Page Down ແມ່ນ fn ພ້ອມລູກສອນຂຶ້ນ ແລະ ລົງ ໃນແປ້ນພິມທີ່ບໍ່ມີປຸ່ມເຫຼົ່ານັ້ນ.
- Backspace ແມ່ນປຸ່ມທີ່ມີປ້າຍວ່າ delete.
:::

:::windows
- {{key:Mod}} ແມ່ນ Ctrl: {{key:Mod+Z}}, {{key:Mod+S}}, {{key:Mod+O}}.
- ເທິງແທຣັກແພດແລັບທັອບ, ການປັດສອງນິ້ວຈະຊູມແຜ່ນ. ເພື່ອເລື່ອນ, ກົດ ແລ້ວລາກ.
:::

:::linux
- {{key:Mod}} ແມ່ນ Ctrl. ບາງ desktop ໃຊ້ທາງລັດ Ctrl+Alt ຫຼື Super ຂອງຕົນເອງ; TakeoffCraft ບໍ່ໃຊ້ອັນໃດໃນນັ້ນ.
- ເທິງແທຣັກແພດ, ການປັດສອງນິ້ວຈະຊູມແຜ່ນ. ເພື່ອເລື່ອນ, ກົດ ແລ້ວລາກ.
:::

:::touch
TakeoffCraft ອ່ານການສຳຜັດເປັນຕົວຊີ້ດຽວ:

- ແຕະເພື່ອວາງຈຸດ, ເລືອກເຄື່ອງໝາຍ ຫຼື ເພີ່ມຈຸດນັບ.
- ລາກດ້ວຍນິ້ວດຽວເພື່ອເລື່ອນແຜ່ນ (ໃນ {{ui:ນັບອັດຕະໂນມັດ}} ມັນແຕ້ມກ່ອງຄົ້ນຫາແທນ).
- ການບີບນິ້ວເພື່ອຊູມ ຍັງບໍ່ຮອງຮັບ. ໃຊ້ {{ui:ພໍດີໜ້າ}} ແລະ {{ui:ພໍດີຄວາມກວ້າງ}}, ຫຼື {{key:Plus}} ແລະ {{key:Minus}} ເທິງແປ້ນພິມ.
- ເພື່ອປິດພື້ນທີ່, ແຕະຈຸດທຳອິດຂອງມັນ. ເພື່ອຈົບຊຸດເສັ້ນ, ແຕະສອງເທື່ອບ່ອນທີ່ browser ຂອງທ່ານປ່ຽນການແຕະສອງເທື່ອເປັນການຄລິກສອງເທື່ອ, ຫຼື ກົດ {{key:Enter}} ເທິງແປ້ນພິມ.
- ຈະບໍ່ມີການອ່ານຄວາມຍາວສົດໆ ຈົນກວ່າທ່ານຈະແຕະ, ເພາະບໍ່ມີການລອຍເທິງຕົວຊີ້.

ໃນປັດຈຸບັນ ແປ້ນພິມ ແລະ ເມົ້າ ຫຼື ແທຣັກແພດ ເປັນວິທີທີ່ດີກວ່າສຳລັບການຖອດປະລິມານ.
:::

## ປຸ່ມທີ່ browser ເກັບໄວ້

ເບຣາເຊີເກັບທາງລັດບາງອັນໄວ້ສຳລັບຕົນເອງ, ແລະ ໜ້າເວັບບໍ່ສາມາດໃຊ້ມັນໄດ້: ປິດແທັບ ({{key:Mod+W}}), ແທັບ ຫຼື ໜ້າຕ່າງໃໝ່ ({{key:Mod+T}}, {{key:Mod+N}}), ແລະ ການຊູມຂອງ browser ({{key:Mod+Plus}}, {{key:Mod+Minus}}). ນັ້ນແມ່ນເຫດຜົນທີ່ TakeoffCraft ຊູມແຜ່ນດ້ວຍລໍ້ ແລະ ດ້ວຍປຸ່ມ {{key:Plus}} ແລະ {{key:Minus}} ດ່ຽວ. ຖ້າທ່ານກົດ {{key:Mod+Plus}} ໂດຍຜິດ, ທັງໜ້າຈະໃຫຍ່ຂຶ້ນ; {{key:Mod+0}} ເອົາມັນກັບຄືນ. ເບິ່ງ [ປຸ່ມ, ເມົ້າ ແລະ ແທຣັກແພດ](common:keys).
