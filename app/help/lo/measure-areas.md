ລາຍການເນື້ອທີ່ວັດເນື້ອທີ່ທີ່ຖືກປິດ. ແຕ່ລະເສັ້ນຮອບສາມາດມີຮູຕັດອອກ (ຊ່ອງແນວຕັ້ງ, ເສົາ, ກະຖາງ), ແລະ ຂອບຂອງມັນສາມາດເປັນເສັ້ນຊື່ ຫຼື ໂຄ້ງໄດ້.

ກ່ອນເລີ່ມ: ແຜ່ນຕ້ອງມີມາດຕາສ່ວນ ([ກຳນົດມາດຕາສ່ວນ](topic:set-the-scale)) ແລະ ທ່ານຕ້ອງມີລາຍການເນື້ອທີ່ ([ສ້າງລາຍການການຖອດປະລິມານ](topic:create-items)).

## ຕີເສັ້ນຮອບເນື້ອທີ່

1. ຄລິກລາຍການເນື້ອທີ່ໃຕ້ {{ui:ລາຍການ}}.
2. ເລືອກ {{ui:ພື້ນທີ່}} ໃນແຖບເຄື່ອງມື, ຫຼື ກົດ {{key:A}}.
3. ຄລິກແຕ່ລະມຸມຂອງຫ້ອງ ຫຼື ແຜ່ນຕາມລຳດັບ. ຖື {{key:Shift}} ຄ້າງໄວ້ສຳລັບຂອບ 0°, 45° ແລະ 90°.
   ຈາກຈຸດທີສາມເປັນຕົ້ນໄປ, ເສັ້ນຮອບຈະຖືກຕື່ມ ແລະ ແຖບສະຖານະສະແດງ "ເນື້ອທີ່" ແລະ "ຄວາມຍາວຮອບ". ວົງແຫວນນ້ອຍປາກົດອ້ອມຈຸດທຳອິດ.
4. ປິດເສັ້ນຮອບ: ຄລິກຈຸດທຳອິດ (ພາຍໃນວົງແຫວນ), ກົດ {{key:Enter}}, ຄລິກສອງເທື່ອ, ຫຼື ຄລິກຂວາ.
   ເນື້ອທີ່ຖືກຕື່ມເປັນສີອ່ອນຕາມສີຂອງລາຍການ, ມີປ້າຍເປັນຕາແມັດຟຸດ, ແລະ ກາຍເປັນເຄື່ອງໝາຍທີ່ເລືອກ.

{{key:Backspace}} ລຶບມຸມສຸດທ້າຍ; {{key:Esc}} ຖິ້ມເສັ້ນຮອບ.

ເສັ້ນຮອບທີ່ຕັດກັບຕົວເອງຈະຖືກປະຕິເສດ: "ເສັ້ນຮອບຕັດກັບຕົວເອງ, ດັ່ງນັ້ນເນື້ອທີ່ຈະຜິດ. ແຕ້ມໃໝ່ໂດຍບໍ່ໃຫ້ຕັດກັນ." ໃຫ້ແຕ້ມຮູບແປດເປັນສອງເນື້ອທີ່ແທນ.

## ຂອບໂຄ້ງ

ຕອນຕີເສັ້ນຮອບ, ເປີດໂໝດເສັ້ນໂຄ້ງ ({{ui:ເສັ້ນໂຄ້ງ}} ຫຼື {{key:R}}). ຫຼັງຈາກມຸມໜຶ່ງ, ຄລິກຈຸດຕາມເສັ້ນໂຄ້ງ, ແລ້ວບ່ອນທີ່ເສັ້ນໂຄ້ງສິ້ນສຸດ. ເພື່ອປິດເສັ້ນຮອບດ້ວຍເສັ້ນໂຄ້ງ, ໃຫ້ຄລິກສຸດທ້າຍນັ້ນເປັນຈຸດທຳອິດ. ເນື້ອທີ່ ແລະ ຄວາມຍາວຮອບໃຊ້ເສັ້ນໂຄ້ງທີ່ແທ້ຈິງ. ກົດ {{key:R}} ອີກເທື່ອເພື່ອກັບໄປເປັນຂອບເສັ້ນຊື່. ເພື່ອປ່ຽນເສັ້ນໂຄ້ງພາຍຫຼັງ, ເລືອກເນື້ອທີ່ ແລະ ລາກຕົວຈັບກົມຢູ່ກາງເສັ້ນໂຄ້ງ. ເບິ່ງ [ວັດເສັ້ນຕໍ່ເນື່ອງ](topic:measure-linear#arcs-and-curved-walls).

## ວົງມົນ

1. ດ້ວຍລາຍການເນື້ອທີ່, ເລືອກ {{ui:ວົງມົນ}} ຫຼື ກົດ {{key:O}}.
2. ຄລິກຈຸດສູນກາງ.
3. ຄລິກຈຸດໃດກໍໄດ້ເທິງຂອບ.
   ຕອນທ່ານເລື່ອນ, ແຖບສະຖານະສະແດງລັດສະໝີ ແລະ ເນື້ອທີ່. ວົງມົນຖືກເພີ່ມເປັນເຄື່ອງໝາຍເນື້ອທີ່ທີ່ມີເນື້ອທີ່ ແລະ ຄວາມຍາວຮອບທີ່ແທ້ຈິງ.

## ເຕີມຫ້ອງດ້ວຍການຄລິກດຽວ

ເມື່ອຫ້ອງຖືກແຕ້ມເປັນຝາປິດ, {{ui:ຫ້ອງ}} ({{key:B}}) ຈະເອົາພື້ນອອກດ້ວຍການຄລິກດຽວ:

1. ເລືອກລາຍການເນື້ອທີ່ (ຕົວຢ່າງ ການຕົກແຕ່ງພື້ນ) ແລະ ໃຫ້ແຜ່ນມີມາດຕາສ່ວນ.
2. ເລືອກ {{ui:ຫ້ອງ}} ໃນກຸ່ມ {{ui:ການຖອດປະລິມານ}}, ຫຼື ກົດ {{key:B}}.
3. ຄລິກພາຍໃນຫ້ອງ, ໃນພື້ນທີ່ເປີດ (ບໍ່ແມ່ນເທິງເສັ້ນ ຫຼື ປ້າຍ).

TakeoffCraft ອ່ານເສັ້ນຂອງແຜ່ນ, ຊອກພື້ນທີ່ອ້ອມຈຸດທີ່ຄລິກ ແລະ ຕີເສັ້ນຮອບຕາມໜ້າດ້ານພາຍໃນຂອງຝາ, ເປັນເນື້ອທີ່ຂອງລາຍການປັດຈຸບັນ. ຊ່ອງປະຕູເຖິງ **4'-0"** ຖືວ່າປິດ, ດັ່ງນັ້ນເສັ້ນຮອບຈະຜ່ານປະຕູແບບຊື່; **{{ui:ການຕັ້ງຄ່າ}} › TakeoffCraft › ເຕີມຫ້ອງປິດຊ່ອງເປີດເຖິງ** ປ່ຽນຄວາມກວ້າງນັ້ນ (3'-0" ຫາ 8'-0"). ຊື່ຫ້ອງ, ເຟີນີເຈີ ແລະ ອຸປະກອນພາຍໃນຫ້ອງບໍ່ເຮັດໃຫ້ເກີດຮູ.

ມັນປະຕິເສດ, ພ້ອມບອກເຫດຜົນ, ເມື່ອ:

- ການຄລິກຢູ່ເທິງເສັ້ນ ("ນັ້ນຢູ່ເທິງເສັ້ນ");
- ພື້ນທີ່ບໍ່ປິດ: ຊ່ອງເປີດກວ້າງກວ່າການຕັ້ງຄ່າ ເຮັດໃຫ້ມັນໄປຮອດຂອບແຜ່ນ. ເພີ່ມການຕັ້ງຄ່າ, ຫຼື ຕີເສັ້ນຮອບຫ້ອງນັ້ນດ້ວຍມືດ້ວຍ {{ui:ພື້ນທີ່}};
- ພື້ນທີ່ນ້ອຍເກີນໄປ (ສັນຍາລັກ, ບໍ່ແມ່ນຫ້ອງ).

ເມື່ອປະຕູກວ້າງກວ່າການຕັ້ງຄ່າ, ການເຕີມຈະໄປເຖິງຫ້ອງຖັດໄປ ແລະ ຕີເສັ້ນຮອບທັງສອງນຳກັນ: ຍ້ອນກັບ ({{key:Mod+Z}}), ເພີ່ມການຕັ້ງຄ່າ, ແລ້ວຄລິກອີກເທື່ອ. ກວດເສັ້ນຮອບຄືກັບເນື້ອທີ່ໃດໆ: ລາກມຸມຂອງມັນເພື່ອແກ້ໄຂ. ແຜ່ນທີ່ສະແກນກໍເຮັດວຽກຄືກັນ, ແຕ່ເສັ້ນຂອງມັນຕ້ອງເຂັ້ມພໍ; ເສັ້ນສີເທົາອ່ອນອາດບໍ່ຢຸດການເຕີມ.

## ຕັດຮູອອກຈາກເນື້ອທີ່

1. ເລືອກ {{ui:ເລືອກ}} ({{key:V}}) ແລ້ວຄລິກເນື້ອທີ່. ມັນສະແດງຕົວຈັບສີ່ຫຼ່ຽມຢູ່ມຸມ.
2. ເລືອກ {{ui:ຕັດຮູ}} ໃນແຖບເຄື່ອງມື (ຫຼື ກົດ {{key:X}}), ຫຼື {{ui:ເພີ່ມ Cut-out}} ພາຍໃຕ້ **ເຄື່ອງໝາຍທີ່ເລືອກ**.
3. ຄລິກມຸມຂອງຮູ, ພາຍໃນເນື້ອທີ່. ໂໝດເສັ້ນໂຄ້ງກໍໃຊ້ໄດ້ທີ່ນີ້.
4. ປິດມັນເໝືອນເນື້ອທີ່: ຄລິກຈຸດທຳອິດ, ກົດ {{key:Enter}}, ຄລິກສອງເທື່ອ ຫຼື ຄລິກຂວາ.
   ຮູຖືກປ່ອຍໃຫ້ວ່າງ, ແລະ ແຖບສະຖານະບອກວ່າ "ເພີ່ມການຕັດຮູແລ້ວ. ແຕ້ມອີກອັນ, ຫຼື ກົດ {{key:Esc}}."
5. ແຕ້ມຮູອີກໃນເນື້ອທີ່ດຽວກັນ, ຫຼື ກົດ {{key:Esc}}.

ສຳລັບຮູກົມ, ເລືອກ {{ui:ເພີ່ມ Cut-out ວົງມົນ}} ພາຍໃຕ້ **ເຄື່ອງໝາຍທີ່ເລືອກ**, ຄລິກຈຸດສູນກາງ, ແລ້ວຈຸດໃດໜຶ່ງເທິງຂອບ.

ການຕັດຮູຕ້ອງຢູ່ພາຍໃນເນື້ອທີ່ທັງໝົດ ແລະ ຕ້ອງບໍ່ທັບກັບການຕັດຮູອື່ນ. ຖ້າບໍ່ດັ່ງນັ້ນ TakeoffCraft ຈະປະຕິເສດມັນ: "ການຕັດຮູຕ້ອງຢູ່ພາຍໃນເນື້ອທີ່ທັງໝົດ." ຫຼື "ການຕັດຮູບໍ່ສາມາດທັບກັນໄດ້."

ເພື່ອເອົາຮູສຸດທ້າຍອອກອີກ, ເລືອກ {{ui:ລຶບ Cut-out ສຸດທ້າຍ}}.

## ການຫັກກັບຄືນ: ເສົາ, ຊ່ອງແນວຕັ້ງ ແລະ ຊ່ອງເປີດທີ່ຫັກອອກຈາກຫຼາຍລາຍການ

ການຕັດຮູເປັນຂອງເຄື່ອງໝາຍເນື້ອທີ່ອັນດຽວ. **ການຫັກກັບຄືນ** ແຕ້ມຄັ້ງດຽວ ແລະ ຖືກຫັກອອກຈາກທຸກເນື້ອທີ່ທີ່ມັນທັບ, ສຳລັບລາຍການທີ່ທ່ານເລືອກ: ແຕ້ມເສົາຄັ້ງດຽວ, ແລະ ພວກມັນຈະຖືກຫັກອອກຈາກການຕົກແຕ່ງພື້ນ, ເພດານ ແລະ ແຜ່ນພື້ນພ້ອມກັນ.

1. ສ້າງລາຍການເນື້ອທີ່ສຳລັບພວກມັນ, ເຊັ່ນ "ເສົາ".
2. ໃນການຕັ້ງຄ່າລາຍການ, ໝາຍ **ຫັກກັບຄືນ: ເຄື່ອງໝາຍຂອງມັນຫັກເນື້ອທີ່ຂອງມັນອອກຈາກລາຍການອື່ນ**. ລາຍການເນື້ອທີ່ອື່ນຂອງທ່ານຈະປາກົດ, ທັງໝົດຖືກໝາຍໄວ້: ໃຫ້ເອົາເຄື່ອງໝາຍອອກສຳລັບອັນທີ່ການຫັກກັບຄືນບໍ່ຄວນແຕະ (ຕົວຢ່າງ ສີທີ່ຝາ).
3. ຖອດປະລິມານເສົາດ້ວຍ {{ui:ພື້ນທີ່}} ຫຼື {{ui:ວົງມົນ}} ຕາມປົກກະຕິ. ແຕ່ລະອັນຖືກແຕ້ມເປັນເສັ້ນປະ.

ທຸກເຄື່ອງໝາຍເນື້ອທີ່ຂອງລາຍການທີ່ໝາຍໄວ້ໃນແຜ່ນດຽວກັນຈະສູນເສຍສ່ວນທີ່ການຫັກກັບຄືນຄອບຄຸມ: ທັງໝົດຂອງການຫັກກັບຄືນເມື່ອມັນຢູ່ພາຍໃນ, ສະເພາະສ່ວນທີ່ທັບກັນເມື່ອມັນຂ້າມຂອບ, ແລະ ບໍ່ມີຫຍັງພາຍໃນການຕັດຮູ. ແຜງຂອງເນື້ອທີ່ທີ່ເລືອກສະແດງ **ການຫັກກັບຄືນ** ແລະ ເນື້ອທີ່ທີ່ເຫຼືອ. ປະລິມານຂອງລາຍການ, ສູດການປະກອບ, ຄ່າໃຊ້ຈ່າຍ, ມຸມມອງກວດ 3D ແລະ ການສົ່ງອອກ ທັງໝົດໃຊ້ເນື້ອທີ່ທີ່ເຫຼືອ; ໝາຍເຫດຂອງ {{ui:ການປະເມີນລາຄາ}} ບອກວ່າການຫັກກັບຄືນຫັກອອກເທົ່າໃດ. ລາຍການການຫັກກັບຄືນເອງນັບເນື້ອທີ່ຂອງມັນເອງ (ປ່ອຍຄ່າໃຊ້ຈ່າຍໄວ້ເປັນສູນ ຖ້າທ່ານບໍ່ຕັ້ງລາຄາ).

ການຫັກກັບຄືນຕ້ອງເປັນ **ນູນອອກ (convex)**: ສີ່ຫຼ່ຽມ, ວົງມົນ ຫຼື ເສັ້ນຮອບໃດໆທີ່ບໍ່ມີມຸມເຂົ້າໃນ. TakeoffCraft ປະຕິເສດການແຕ້ມອັນທີ່ບໍ່ແມ່ນ; ໃຫ້ແຕ້ມຊ່ອງແນວຕັ້ງຮູບ L ເປັນສອງການຫັກກັບຄືນ. ການຫັກກັບຄືນບໍ່ສາມາດມີການຕັດຮູ, ແລະ ການຫັກກັບຄືນສອງອັນທີ່ທັບກັນນັບທັງສອງ.

ຄວາມຍາວຮອບ (`P`) ບໍ່ປ່ຽນ: ການຫັກກັບຄືນຫັກເນື້ອທີ່ເທົ່ານັ້ນ.

## ອ່ານເນື້ອທີ່

ເລືອກເນື້ອທີ່. **ເຄື່ອງໝາຍທີ່ເລືອກ** ສະແດງ:

| ແຖວ | ຄວາມໝາຍ |
|---|---|
| ເນື້ອທີ່ | ເນື້ອທີ່ສຸດທິ: ເສັ້ນຮອບລົບການຕັດຮູ. ນີ້ແມ່ນສິ່ງທີ່ລາຍການນັບ. |
| ເສັ້ນຮອບ | ເນື້ອທີ່ລວມພາຍໃນເສັ້ນຮອບ (ສະແດງເມື່ອມີການຕັດຮູ). |
| ການຕັດຮູ | ມີຮູຈັກຮູ, ແລະ ເນື້ອທີ່ລວມຂອງພວກມັນ. |
| ຄວາມຍາວຮອບ | ຄວາມຍາວຂອບເສັ້ນຮອບ, **ບວກ** ຂອບຂອງທຸກການຕັດຮູ ("ລວມການຕັດຮູ"). |

ຄວາມຍາວຮອບລວມຂອບຂອງການຕັດຮູໂດຍເຈດຕະນາ, ສຳລັບສິ່ງຕ່າງໆເຊັ່ນ ແບບຂອບ ຫຼື ຖານທີ່ວົນຮອບຮູນຳ. ຖ້າທ່ານຕ້ອງການພຽງແຕ່ຂອບນອກ, ໃຫ້ຮັກສາຮູອອກຈາກລາຍການນັ້ນ ຫຼື ວັດຂອບນອກເປັນຊຸດເສັ້ນ.

{{ui:ຫົວໜ່ວຍ}} ຂອງລາຍການກຳນົດວ່າຜົນລວມຈະຖືກລາຍງານແນວໃດ: SF, SY ຫຼື m². ແຜງ **ເຄື່ອງໝາຍທີ່ເລືອກ** ແລະ ແຖບສະຖານະສະແດງຕາແມັດຟຸດສະເໝີ.

## ແກ້ໄຂເນື້ອທີ່

- **ຍ້າຍມຸມ**: ເລືອກເນື້ອທີ່ ແລ້ວລາກຕົວຈັບມຸມ (ຮູກໍມີຕົວຈັບ). ຖ້າຮູບຮ່າງໃໝ່ຕັດກັນເອງ ຫຼື ດັນຮູອອກນອກເສັ້ນ, ມຸມຈະຖືກວາງກັບຄືນ: "ຈຸດຖືກວາງກັບຄືນແລ້ວ."
- **ຍ້າຍໄປລາຍການອື່ນ**: ເລືອກລາຍການເນື້ອທີ່ອື່ນໃນລາຍການ **ລາຍການ** ພາຍໃຕ້ **ເຄື່ອງໝາຍທີ່ເລືອກ**.
- **ລຶບ**: {{key:Delete}}, {{ui:ລຶບ}} ໃນເມນູ {{ui:ເພີ່ມເຕີມ}} ຂອງແຖບເຄື່ອງມື, ຫຼື {{ui:ລຶບເຄື່ອງໝາຍ}}.
